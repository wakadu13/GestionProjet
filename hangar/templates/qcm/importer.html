Format d’un set (exemple minimal, types de questions) 
{
  "titre": "Mon premier set",
  "description": "Facultatif",
  "questions": [
    {"type": "QCM", "question": "2 + 2 = ?", "choix": ["3", "4", "5"], "reponses": [1]},
    {"type": "QCR", "question": "Nombres pairs ?", "choix": ["2", "3", "4"], "reponses": [0, 2]},
    {"type": "TROU", "question": "La capitale de la France est ___.", "reponses_texte": ["Paris"]},
    {"type": "QRC", "question": "chien", "consigne": "Traduire en anglais", "reponses_texte": ["dog"], "inverse": true},
    {"type": "FLASHCARD", "question": "$E = mc^2$ ?", "verso": "Équivalence masse-énergie", "explication": "…"}
  ]
}
